435:    def propose(
436:        self,
437:        # [num_tokens]
438:        target_token_ids: torch.Tensor,
439:        # [num_tokens] or [3, num_tokens] when M-RoPE is enabled
440:        target_positions: torch.Tensor,
441:        # [num_tokens, hidden_size]
442:        target_hidden_states: torch.Tensor,
443:        # [batch_size]
444:        next_token_ids: torch.Tensor,
445:        token_indices_to_sample: torch.Tensor | None,
446:        common_attn_metadata: CommonAttentionMetadata,
447:        sampling_metadata: SamplingMetadata,
448:        mm_embed_inputs: tuple[list[torch.Tensor], torch.Tensor] | None = None,
449:        num_rejected_tokens_gpu: torch.Tensor | None = None,
450:        slot_mappings: dict[str, torch.Tensor]
451:        | list[dict[str, torch.Tensor]]
452:        | None = None,
453:    ) -> torch.Tensor:
454:        self._last_draft_probs = None
455:        batch_size = common_attn_metadata.batch_size()
456:
457:        if self.method in ("eagle3", "dflash"):
458:            assert isinstance(
459:                self.model,
460:                (
461:                    Eagle3LlamaForCausalLM,
462:                    Eagle3DeepseekV2ForCausalLM,
463:                    DFlashQwen3ForCausalLM,
464:                ),
465:            )
466:            target_hidden_states = self.model.combine_hidden_states(
467:                target_hidden_states
468:            )
469:            assert target_hidden_states.shape[-1] == self.hidden_size
470:
471:        num_tokens, token_indices_to_sample, common_attn_metadata = (
472:            self.set_inputs_first_pass(
473:                target_token_ids=target_token_ids,
474:                next_token_ids=next_token_ids,
475:                target_positions=target_positions,
476:                target_hidden_states=target_hidden_states,
477:                token_indices_to_sample=token_indices_to_sample,
478:                cad=common_attn_metadata,
479:                num_rejected_tokens_gpu=num_rejected_tokens_gpu,
480:            )
481:        )
482:
483:        per_group_attn_metadata, per_layer_attn_metadata = (
484:            self.build_per_group_and_layer_attn_metadata(common_attn_metadata)
485:        )
486:
487:        cudagraph_runtime_mode, num_input_tokens, num_tokens_across_dp = (
488:            self._determine_batch_execution_and_padding(num_tokens)
489:        )
490:
491:        model_kwargs, slot_mapping_size = self.build_model_inputs_first_pass(
492:            num_tokens, num_input_tokens, mm_embed_inputs
493:        )
494:        # Step 0 of index_share_for_mtp_iteration: let the MTP layer
495:        # compute its own indices (skip_topk=False) so subsequent steps
496:        # can reuse them.
497:        if self._share_mtp_indices and hasattr(self.model.model, "set_skip_topk"):
498:            self.model.model.set_skip_topk(False)
499:
500:        with set_forward_context(
501:            per_layer_attn_metadata,
502:            self.vllm_config,
503:            num_tokens=num_input_tokens,
504:            num_tokens_across_dp=num_tokens_across_dp,
505:            cudagraph_runtime_mode=cudagraph_runtime_mode,
506:            slot_mapping=self._get_slot_mapping(
507:                slot_mapping_size, common_attn_metadata.slot_mapping
508:            ),
509:        ):
510:            ret_hidden_states = self.model(**model_kwargs)
511:            if not self.model_returns_tuple():
512:                last_hidden_states = ret_hidden_states
513:                hidden_states = last_hidden_states
514:            else:
515:                last_hidden_states, hidden_states = ret_hidden_states
516:
517:        # After step 0: switch to reuse mode so steps 1+ skip the indexer
518:        # and read the indices that step 0 just wrote into the shared buffer.
519:        if self._share_mtp_indices and hasattr(self.model.model, "set_skip_topk"):
520:            self.model.model.set_skip_topk(True)
521:
522:        sample_hidden_states = last_hidden_states[token_indices_to_sample]
523:
524:        # Early exit if there is only one draft token to be generated.
525:        if self.num_speculative_tokens == 1 or self.parallel_drafting:
526:            draft_token_ids, draft_probs = self._sample_draft_tokens(
527:                sample_hidden_states, sampling_metadata
528:            )
529:            if draft_probs is not None:
530:                self._last_draft_probs = draft_probs.view(
531:                    -1, self.num_speculative_tokens, draft_probs.shape[-1]
532:                ).contiguous()
533:            return draft_token_ids.view(-1, self.num_speculative_tokens)
534:
535:        if self.uses_mrope:
536:            positions = self.mrope_positions[:, token_indices_to_sample]
537:        else:
538:            positions = self.positions[token_indices_to_sample]
539:        hidden_states = hidden_states[token_indices_to_sample]
540:
541:        if self.constant_draft_positions:
542:            # Write the sampling positions into the front of the
543:            # positions buffer so that subsequent loop iterations
544:            # (which read via _get_positions) use the correct values.
545:            self.positions[:batch_size] = positions
546:
547:        draft_token_ids, draft_probs = self._sample_draft_tokens(
548:            sample_hidden_states, sampling_metadata
549:        )
550:        draft_probs_list = None if draft_probs is None else [draft_probs]
551:
552:        if self.allowed_attn_types is not None:
553:            for group_md in per_group_attn_metadata:
554:                if not isinstance(group_md, self.allowed_attn_types):
555:                    raise ValueError(
556:                        f"Unsupported attention metadata type for speculative "
557:                        "decoding with num_speculative_tokens > 1: "
558:                        f"{type(group_md)}. Supported types are: "
559:                        f"{self.allowed_attn_types}"
560:                    )
561:
562:        # Generate the remaining draft tokens.
563:        draft_token_ids_list = [draft_token_ids]
564:
565:        cudagraph_runtime_mode, input_batch_size, batch_size_across_dp = (
566:            self._determine_batch_execution_and_padding(batch_size)
567:        )
568:
569:        common_attn_metadata.num_actual_tokens = batch_size
570:        common_attn_metadata.max_query_len = 1
571:        common_attn_metadata.query_start_loc = self.arange[: batch_size + 1]
572:        common_attn_metadata.query_start_loc_cpu = torch.from_numpy(
573:            self.token_arange_np[: batch_size + 1]
574:        ).clone()
575:
576:        # In padded drafter batch, we need to adjust the sequence lengths
577:        # to remove the "padding" (i.e. rejected tokens).
578:        # Only apply this adjustment when we have rejected tokens
579:        # (i.e., not the first proposal).
580:        if self.num_speculative_tokens > 1 and num_rejected_tokens_gpu is not None:
581:            common_attn_metadata.seq_lens -= num_rejected_tokens_gpu
582:            # Invalidate the CPU-side shadows to avoid H<>D sync.
583:            common_attn_metadata._seq_lens_cpu = None
584:            common_attn_metadata._num_computed_tokens_cpu = None
585:
586:        block_size = self.block_size
587:        assert block_size > 0, "block_size has not been initialized."
588:        for token_index in range(self.num_speculative_tokens - 1):
589:            # Update the inputs.
590:            # cast to int32 is crucial when eagle model is compiled.
591:            # tensor.argmax() returns int64 by default.
592:            input_ids = draft_token_ids_list[-1].int()
593:
594:            if not self.constant_draft_positions:
595:                positions = self._update_positions_dependent_metadata(
596:                    positions,
597:                    common_attn_metadata,
598:                    batch_size,
599:                    input_batch_size,
600:                    block_size,
601:                )
602:
603:            # Rebuild attention metadata. When draft positions are constant
604:            # (e.g. Gemma4 MTP), common_attn_metadata is invariant across
605:            # loop iterations so we build once and reuse.
606:            if not self.constant_draft_positions or token_index == 0:
607:                _, per_layer_attn_metadata = (
608:                    self.build_per_group_and_layer_attn_metadata(
609:                        common_attn_metadata, draft_index=token_index + 1
610:                    )
611:                )
612:
613:            # copy inputs to buffer for cudagraph
614:            self.input_ids[:batch_size] = input_ids
615:            self.hidden_states[:batch_size] = hidden_states
616:            if self.supports_mm_inputs:
617:                self.inputs_embeds[:batch_size] = self.model.embed_input_ids(input_ids)
618:
619:                input_ids = None
620:                inputs_embeds = self.inputs_embeds[:input_batch_size]
621:            else:
622:                input_ids = self.input_ids[:input_batch_size]
623:                inputs_embeds = None
624:
625:            # Run the model.
626:            model_kwargs = {
627:                "input_ids": input_ids,
628:                "positions": self._get_positions(input_batch_size),
629:                "inputs_embeds": inputs_embeds,
630:            }
631:            if self.pass_hidden_states_to_model:
632:                model_kwargs["hidden_states"] = self.hidden_states[:input_batch_size]
633:
634:            with set_forward_context(
635:                per_layer_attn_metadata,
636:                self.vllm_config,
637:                num_tokens=input_batch_size,
638:                num_tokens_across_dp=batch_size_across_dp,
639:                cudagraph_runtime_mode=cudagraph_runtime_mode,
640:                slot_mapping=self._get_slot_mapping(input_batch_size),
641:            ):
642:                ret_hidden_states = self.model(**model_kwargs)
643:                if not self.model_returns_tuple():
644:                    last_hidden_states = ret_hidden_states
645:                    hidden_states = ret_hidden_states
646:                else:
647:                    last_hidden_states, hidden_states = ret_hidden_states
648:
649:            hidden_states = hidden_states[:batch_size]
650:            draft_token_ids, draft_probs = self._sample_draft_tokens(
651:                last_hidden_states[:batch_size], sampling_metadata
652:            )
653:            if draft_probs is not None:
654:                assert draft_probs_list is not None
655:                draft_probs_list.append(draft_probs)
656:            draft_token_ids_list.append(draft_token_ids)
657:
658:        # [batch_size, num_speculative_tokens]
659:        draft_token_ids = torch.stack(draft_token_ids_list, dim=1)
660:        if draft_probs_list is not None:
661:            self._last_draft_probs = torch.stack(draft_probs_list, dim=1).contiguous()
662:        return draft_token_ids

664:    def _update_positions_dependent_metadata(
665:        self,
666:        positions: torch.Tensor,
667:        common_attn_metadata,
668:        batch_size: int,
669:        input_batch_size: int,
670:        block_size: int,
671:    ) -> torch.Tensor:
672:        """Update positions, slot mappings, and sequence metadata for the
673:        next draft step. Returns the updated positions tensor."""
674:        positions_1d = positions[0] if self.uses_mrope else positions
675:        if self.uses_mrope:
676:            out_pos = self.mrope_positions[0, :batch_size]
677:        elif self.uses_xdrope_dim > 0 and self.draft_uses_xdrope_dim > 0:
678:            out_pos = self.xdrope_positions[0, :batch_size]
679:        else:
680:            out_pos = self.positions[:batch_size]
681:        eagle_step_update_slot_mapping_and_metadata(
682:            positions_1d=positions_1d,
683:            block_table_tensor=common_attn_metadata.block_table_tensor,
684:            seq_lens=common_attn_metadata.seq_lens,
685:            block_size=block_size,
686:            max_model_len=self.max_model_len,
687:            out_clamped_positions=out_pos,
688:            out_slot_mapping=self._slot_mapping_buffer[:input_batch_size],
689:            input_batch_size=input_batch_size,
690:        )
691:        common_attn_metadata.slot_mapping = self._slot_mapping_buffer[:batch_size]
692:        if self.uses_mrope:
693:            self.mrope_positions[1:, :batch_size] = self.mrope_positions[0, :batch_size]
694:            positions = self.mrope_positions[:, :batch_size]
695:        elif self.uses_xdrope_dim > 0 and self.draft_uses_xdrope_dim > 0:
696:            self.xdrope_positions[1:, :batch_size] = self.xdrope_positions[
697:                0, :batch_size
698:            ]
699:            positions = self.xdrope_positions[0, :batch_size]
700:        else:
701:            positions = self.positions[:batch_size]
702:        common_attn_metadata.max_seq_len = min(
703:            common_attn_metadata.max_seq_len + 1,
704:            self.max_model_len,
705:        )
706:
707:        if common_attn_metadata._seq_lens_cpu is not None:
708:            common_attn_metadata._seq_lens_cpu += 1
709:        if common_attn_metadata._num_computed_tokens_cpu is not None:
710:            common_attn_metadata._num_computed_tokens_cpu += 1
711:        if common_attn_metadata.seq_lens_cpu_upper_bound is not None:
712:            common_attn_metadata.seq_lens_cpu_upper_bound += 1
713:
714:        return positions

716:    def set_inputs_first_pass(
717:        self,
718:        target_token_ids: torch.Tensor,
719:        next_token_ids: torch.Tensor,
720:        target_positions: torch.Tensor,
721:        target_hidden_states: torch.Tensor,
722:        token_indices_to_sample: torch.Tensor | None,
723:        cad: CommonAttentionMetadata,
724:        num_rejected_tokens_gpu: torch.Tensor | None,
725:    ) -> tuple[int, torch.Tensor, CommonAttentionMetadata]:
726:        if not self.needs_extra_input_slots:
727:            # Default EAGLE pathway: no reshaping of input tensors needed.
728:            # Simply rotate the input ids and leave the positions unchanged,
729:            # Inserting the next token ids at the last slot in each request.
730:            if token_indices_to_sample is None:
731:                token_indices_to_sample = cad.query_start_loc[1:] - 1
732:
733:            num_tokens = target_token_ids.shape[0]
734:            # Shift the input ids by one token.
735:            # E.g., [a1, b1, b2, c1, c2, c3] -> [b1, b2, c1, c2, c3, c3]
736:            self.input_ids[: num_tokens - 1] = target_token_ids[1:]
737:            # Replace the last token with the next token.
738:            # E.g., [b1, b2, c1, c2, c3, c3] -> [a2, b2, b3, c2, c3, c4]
739:            self.input_ids[token_indices_to_sample] = next_token_ids
740:
741:            # copy inputs to buffer for cudagraph
742:            if self.uses_xdrope_dim > 0 and self.draft_uses_xdrope_dim == 0:
743:                target_positions = target_positions[0]
744:            self._set_positions(num_tokens, target_positions)
745:
746:            self.hidden_states[:num_tokens] = target_hidden_states
747:
748:            return num_tokens, token_indices_to_sample, cad
749:        else:
750:            assert self.is_rejected_token_mask is not None
751:            assert self.is_masked_token_mask is not None
752:            # 1.
753:            # Call a custom triton kernel to copy input_ids and positions
754:            # into the correct slots in the preallocated buffers self.input_ids,
755:            # self.positions.
756:            batch_size = cad.batch_size()
757:            # Since we might have to copy a lot of data for prefills, we select the
758:            # block size based on the max query length and limit to max 256 slots/block.
759:            max_num_tokens_per_request = (
760:                cad.max_query_len + self.net_num_new_slots_per_request
761:            )
762:            BLOCK_SIZE_TOKENS = min(256, next_power_of_2(max_num_tokens_per_request))
763:            num_blocks = (
764:                max_num_tokens_per_request + BLOCK_SIZE_TOKENS - 1
765:            ) // BLOCK_SIZE_TOKENS
766:            total_num_input_tokens = target_token_ids.shape[0]
767:            total_num_output_tokens = total_num_input_tokens + (
768:                self.net_num_new_slots_per_request * batch_size
769:            )
770:
771:            token_indices_to_sample = torch.empty(
772:                batch_size * self.extra_slots_per_request,
773:                dtype=torch.int32,
774:                device=self.device,
775:            )
776:
777:            # Destination indices to write target_hidden_states into drafting buffer.
778:            out_hidden_state_mapping = torch.empty(
779:                total_num_input_tokens, dtype=torch.int32, device=self.device
780:            )
781:
782:            # Kernel grid: one program per request (row)
783:            grid = (batch_size, num_blocks)
784:            query_start_loc = cad.query_start_loc
785:            query_end_loc = cad.query_start_loc[1:] - 1
786:            if num_rejected_tokens_gpu is not None:
787:                query_end_loc = query_end_loc - num_rejected_tokens_gpu
788:
789:            copy_and_expand_eagle_inputs_kernel[grid](
790:                # (Padded) Inputs from the target model
791:                target_token_ids_ptr=target_token_ids,
792:                target_positions_ptr=target_positions,
793:                next_token_ids_ptr=next_token_ids,  # sampled tokens, one per request
794:                # Outputs to the drafting buffers
795:                out_input_ids_ptr=self.input_ids,
796:                out_positions_ptr=self.positions,  # Doesn't support mrope for now
797:                out_is_rejected_token_mask_ptr=self.is_rejected_token_mask,
798:                out_is_masked_token_mask_ptr=self.is_masked_token_mask,
799:                out_new_token_indices_ptr=token_indices_to_sample,
800:                out_hidden_state_mapping_ptr=out_hidden_state_mapping,
801:                # Input metadata
802:                query_start_loc_ptr=query_start_loc,
803:                query_end_loc_ptr=query_end_loc,
804:                padding_token_id=0,
805:                parallel_drafting_token_id=self.parallel_drafting_token_id,
806:                # Sizing info
807:                # Note that we can deduce batch_size for free from the grid size
808:                total_input_tokens=total_num_input_tokens,
809:                num_padding_slots_per_request=self.extra_slots_per_request,
810:                shift_input_ids=self.pass_hidden_states_to_model,
811:                BLOCK_SIZE_TOKENS=BLOCK_SIZE_TOKENS,
812:            )
813:            if self.pass_hidden_states_to_model:
814:                assert self.parallel_drafting_hidden_state_tensor is not None
815:                self.hidden_states[out_hidden_state_mapping] = target_hidden_states
816:                # Use torch.where to avoid DtoH sync from boolean indexing
817:                mask = self.is_masked_token_mask[:total_num_output_tokens]
818:                torch.where(
819:                    mask.unsqueeze(1),
820:                    self.parallel_drafting_hidden_state_tensor,
821:                    self.hidden_states[:total_num_output_tokens],
822:                    out=self.hidden_states[:total_num_output_tokens],
823:                )
824:
825:            # 2.
826:            # Recompute the slot mapping based on the new positions and
827:            # rejection mask.
828:            assert self.block_size > 0, "block_size has not been initialized."
829:            new_slot_mapping = compute_new_slot_mapping(
830:                cad=cad,
831:                new_positions=self.positions[:total_num_output_tokens],
832:                is_rejected_token_mask=self.is_rejected_token_mask[
833:                    :total_num_output_tokens
834:                ],
835:                block_size=self.block_size,
836:                num_new_tokens=self.net_num_new_slots_per_request,
837:                max_model_len=self.max_model_len,
838:            )
839:
840:            # 3. Update the common attention metadata with the new (meta)data
841:            new_cad = extend_all_queries_by_N(
842:                cad,
843:                N=self.net_num_new_slots_per_request,
844:                arange=self.arange,
845:                new_slot_mapping=new_slot_mapping,
846:            )
847:
848:            return total_num_output_tokens, token_indices_to_sample, new_cad

898:    def prepare_next_token_ids_cpu(
899:        self,
900:        sampled_token_ids: list[list[int]],
901:        requests: dict[str, CachedRequestState],
902:        gpu_input_batch: InputBatch,
903:        num_scheduled_tokens: dict[str, int],
904:    ) -> torch.Tensor:
905:        """
906:        This function is used to prepare the inputs for speculative decoding.
907:        It calculates the next token ids for each request based on the sampled
908:        token ids from the CPU. If a request has no sampled token ids (e.g.,
909:        during the initial decoding steps), it falls back to using the request
910:        state to get the next token id.
911:        """
912:        req_ids = gpu_input_batch.req_ids
913:        next_token_ids: list[int] = []
914:        for i, token_ids in enumerate(sampled_token_ids):
915:            if token_ids:
916:                # Common case.
917:                next_token_id = token_ids[-1]
918:            else:
919:                # Partial prefill (rare case).
920:                # Get the next token id from the request state.
921:                req_id = req_ids[i]
922:                req_state = requests[req_id]
923:                seq_len = req_state.num_computed_tokens + num_scheduled_tokens[req_id]
924:                next_token_id = req_state.get_token_id(seq_len)
925:            next_token_ids.append(next_token_id)
926:        next_token_ids = torch.tensor(
927:            next_token_ids, dtype=torch.int32, device=self.input_ids.device
928:        )
929:        return next_token_ids

931:    def prepare_next_token_ids_padded(
932:        self,
933:        sampled_token_ids: torch.Tensor,
934:        requests: dict[str, CachedRequestState],
935:        gpu_input_batch: InputBatch,
936:        discard_request_mask: torch.Tensor,
937:    ) -> tuple[torch.Tensor, torch.Tensor]:
938:        """
939:        This function is used to prepare the inputs for speculative decoding.
940:        It calculates the next token ids and the number of valid sampled tokens
941:        for each request, considering the "discarded" requests whose next token
942:        is not sampled and comes from `request.get_token_id()` instead. This is denoted
943:        the "backup" token id. It also counts rejected tokens via `sampled_token_ids`.
944:        """
945:        # Precompute backup token IDs for discarded requests.
946:        num_reqs = gpu_input_batch.num_reqs
947:        for i in range(num_reqs):
948:            self.backup_next_token_ids.np[i] = requests[
949:                gpu_input_batch.req_ids[i]
950:            ].get_token_id(gpu_input_batch.num_tokens_no_spec[i] - 1)
951:        self.backup_next_token_ids.copy_to_gpu(num_reqs)
952:        backup_tokens_gpu = self.backup_next_token_ids.gpu
953:
954:        batch_size, num_tokens = sampled_token_ids.shape
955:        device = sampled_token_ids.device
956:
957:        assert discard_request_mask.dtype == torch.bool
958:        assert backup_tokens_gpu.dtype == torch.int32
959:
960:        next_token_ids = torch.empty(batch_size, dtype=torch.int32, device=device)
961:        valid_sampled_tokens_count = next_token_ids.new_empty(batch_size)
962:
963:        # Kernel grid: one program per request (row)
964:        grid = (batch_size,)
965:
966:        # Find the next power of 2 for block sizes
967:        BLOCK_SIZE_TOKENS = next_power_of_2(num_tokens)
968:        eagle_prepare_next_token_padded_kernel[grid](
969:            sampled_token_ids,
970:            discard_request_mask,
971:            backup_tokens_gpu,
972:            next_token_ids,
973:            valid_sampled_tokens_count,
974:            gpu_input_batch.vocab_size,
975:            num_tokens,
976:            batch_size,
977:            sampled_token_ids.stride(0),
978:            BLOCK_SIZE_TOKENS=BLOCK_SIZE_TOKENS,
979:        )
980:
981:        return next_token_ids, valid_sampled_tokens_count

983:    def prepare_inputs_padded(
984:        self,
985:        common_attn_metadata: CommonAttentionMetadata,
986:        spec_decode_metadata: SpecDecodeMetadata,
987:        valid_sampled_tokens_count: torch.Tensor,
988:    ) -> tuple[CommonAttentionMetadata, torch.Tensor, torch.Tensor]:
989:        """
990:        This function is used to prepare the inputs for speculative decoding
991:        It updates the common_attn_metadata for speculative decoding,
992:        but does not consider the rejected tokens. Instead, all tokens
993:        are included as inputs to the speculator, with the rejected tokens
994:        used as padding and filtered out later by `token_indices_to_sample`.
995:        No blocking CPU operations should be introduced in this function.
996:        """
997:        num_reqs = common_attn_metadata.num_reqs
998:        device = valid_sampled_tokens_count.device
999:
1000:        token_indices_to_sample = torch.empty(
1001:            (num_reqs,), dtype=torch.int32, device=device
1002:        )
1003:        num_rejected_tokens_gpu = torch.empty(
1004:            (num_reqs,), dtype=torch.int32, device=device
1005:        )
1006:
1007:        grid = (num_reqs,)
1008:        eagle_prepare_inputs_padded_kernel[grid](
1009:            spec_decode_metadata.cu_num_draft_tokens,
1010:            valid_sampled_tokens_count,
1011:            common_attn_metadata.query_start_loc,
1012:            token_indices_to_sample,
1013:            num_rejected_tokens_gpu,
1014:            num_reqs,

1045:    def prepare_inputs(
1046:        self,
1047:        common_attn_metadata: CommonAttentionMetadata,
1048:        sampled_token_ids: list[list[int]],
1049:        num_draft_tokens: list[int],
1050:    ) -> tuple[CommonAttentionMetadata, torch.Tensor]:
1051:        """
1052:        This function is used to prepare the inputs for speculative decoding.
1053:        It updates to the common_attn_metadata to account for the rejected
1054:        tokens (and newly sampled tokens). It also returns the token indices
1055:        of the tokens that should be fed to the speculator.
1056:        """
1057:        # E.g.
1058:        #  common_attn_metadata.query_start_loc{_cpu}:
1059:        #       [0, q1, q1 + q2, q1 + q2 + q3]
1060:        #  common_attn_metadata.seq_lens{_cpu}: [s1, s2, s3]
1061:        #  num_rejected_tokens: [n1, n2, n3]
1062:        # This function computes the intermediate values:
1063:        #  num_tokens_per_req: [q1 - n1, q2 - n2, q3 - n3]
1064:        # And returns:
1065:        #  common_attn_metadata.query_start_loc{_cpu}:
1066:        #       [0, q1 - n1, q1 + q2 - n1 - n2, q1 + q2 + q3 - n1 - n2 - n3]
1067:        #  common_attn_metadata.seq_lens{_cpu}:
1068:        #       [s1 - n1 + 1, s2 - n2 + 1, s3 - n3 + 1]
1069:        #  token_indices: [0, 1, ..., q1 - n1 - 1,
1070:        #                 q1, q1 + 1, ..., q1 + q2 - n2 - 1,
1071:        #                 q1 + q2, q1 + q2 + 1, ..., q1 + q2 + q3 - n3 - 1]
1072:
1073:        num_rejected_tokens = [
1074:            n + 1 - len(sampled_token_ids[i]) if n > 0 else 0
1075:            for i, n in enumerate(num_draft_tokens)
1076:        ]
1077:        num_rejected_tokens = torch.tensor(num_rejected_tokens, dtype=torch.int32)
1078:
1079:        device = common_attn_metadata.query_start_loc.device
1080:        query_start_loc_cpu = common_attn_metadata.query_start_loc_cpu
1081:        # upper_bound - rejected = actual post-rejection seq_lens (no D2H sync).
1082:        assert common_attn_metadata.seq_lens_cpu_upper_bound is not None
1083:        new_seq_lens_cpu = (
1084:            common_attn_metadata.seq_lens_cpu_upper_bound - num_rejected_tokens
1085:        )
1086:
1087:        # [0, q1, q1 + q2, q1 + q2 + q3] -> [q1, q2, q3]
1088:        new_query_len_per_req = query_start_loc_cpu[1:] - query_start_loc_cpu[:-1]
1089:        # [q1, q2, q3] -> [q1 - n1, q2 - n2, q3 - n3]
1090:        new_num_tokens_per_req = new_query_len_per_req - num_rejected_tokens
1091:        new_num_tokens_per_req_np = new_num_tokens_per_req.numpy()
1092:
1093:        # [q1 - n1, q2 - n2, q3 - n3] ->
1094:        # [0, q1 - n1, q1 + q2 - n1 - n2, q1 + q2 + q3 - n1 - n2 - n3]
1095:        new_query_start_loc_cpu = torch.zeros(
1096:            query_start_loc_cpu.shape,
1097:            dtype=torch.int32,
1098:            pin_memory=is_pin_memory_available(),
1099:        )
1100:        new_query_start_loc_np = new_query_start_loc_cpu.numpy()
1101:        np.cumsum(new_num_tokens_per_req_np, out=new_query_start_loc_np[1:])
1102:
1103:        total_num_tokens = new_query_start_loc_np[-1]
1104:        # Example assuming num_tokens_per_req_np = [2, 4, 3]
1105:        # this implies that `new_query_start_locs` is:
1106:        # [0, 2, 6, 9] ->
1107:        # [0, 0, 2, 2, 2, 2, 6, 6, 6]
1108:        #  _r1_  ____r2____  ___r3__
1109:        new_query_start_locs_expanded = np.repeat(
1110:            new_query_start_loc_np[:-1], new_num_tokens_per_req_np
1111:        )
1112:        # [0, 1, 2, 3, 4, 5, 6, 7, 8] ->
1113:        # [0, 1, 0, 1, 2, 3, 0, 1, 2]
1114:        #  _r1_  ____r2____  ___r3__
1115:        token_offsets = (
1116:            self.token_arange_np[:total_num_tokens] - new_query_start_locs_expanded
1117:        )
1118:
1119:        # Expand starting positions to match token pattern
1120:        # [0, q1, q1 + q2] ->
1121:        # [0, 0, q1, q1, q1, q1, q1 + q2, q1 + q2, q1 + q2]
1122:        #  _r1_  _____r2_______  ___________r3____________
1123:        old_query_start_locs_expanded = np.repeat(
1124:            query_start_loc_cpu[:-1].numpy(), new_num_tokens_per_req_np
1125:        )
1126:        # Final token indices are:
1127:        # [0, 1,                                // req 1
1128:        #  q1 + 0, q1 + 1, q1 + 2, q1 + 3,       // req 2
1129:        #  q1 + q2 + 0, q1 + q2 + 1, q1 + q2 + 2] // req 3
1130:        token_indices_np = token_offsets + old_query_start_locs_expanded
1131:        token_indices = torch.from_numpy(token_indices_np).to(device, non_blocking=True)
1132:
1133:        spec_common_attn_metadata = CommonAttentionMetadata(
1134:            query_start_loc=new_query_start_loc_cpu.to(device, non_blocking=True),
1135:            seq_lens=new_seq_lens_cpu.to(device, non_blocking=True),
1136:            query_start_loc_cpu=new_query_start_loc_cpu,
1137:            _seq_lens_cpu=new_seq_lens_cpu,
1138:            _num_computed_tokens_cpu=common_attn_metadata._num_computed_tokens_cpu,
1139:            seq_lens_cpu_upper_bound=new_seq_lens_cpu,
1140:            num_reqs=common_attn_metadata.num_reqs,
1141:            num_actual_tokens=total_num_tokens,
1142:            max_query_len=new_query_len_per_req.max().item(),
1143:            max_seq_len=new_seq_lens_cpu.max().item(),
1144:            block_table_tensor=common_attn_metadata.block_table_tensor,
1145:            slot_mapping=common_attn_metadata.slot_mapping[token_indices],
1146:            causal=True,
1147:            dcp_local_seq_lens=common_attn_metadata.dcp_local_seq_lens,
1148:        )
1149:
1150:        return spec_common_attn_metadata, token_indices

1382:                and isinstance(self.model.lm_head.weight, torch.Tensor)

1386:                    target_language_model.lm_head.weight.cpu(),
1387:                    self.model.lm_head.weight.cpu(),

1400:        else:

1402:            share_lm_head = True

184:    def uninstall_static_kernel(self):
185:        import fcntl
186:        import os
187:        import subprocess
188:
189:        ascend_home_path = os.environ["ASCEND_HOME_PATH"]
190:        static_kernel_dir_path = os.path.join(ascend_home_path, "opp/static_kernel")
191:        uninstall_script_path = os.path.join(static_kernel_dir_path, "ai_core/uninstall.sh")
192:        lock_file_path = os.path.join(static_kernel_dir_path, "uninstall.lock")
193:
194:        if not os.path.exists(uninstall_script_path):
195:            return
196:        with open(lock_file_path, "w") as lock_fd:
197:            try:
198:                fcntl.flock(lock_fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
199:                subprocess.Popen(
200:                    ["bash", uninstall_script_path],
201:                    stdin=subprocess.DEVNULL,
202:                    stdout=subprocess.DEVNULL,
203:                    stderr=subprocess.DEVNULL,
204:                    start_new_session=True,
205:                )
206:            except (BlockingIOError, OSError):
207:                return
208:            finally:
209:                try:
210:                    fcntl.flock(lock_fd, fcntl.LOCK_UN)
211:                    if os.path.exists(lock_file_path):
212:                        os.remove(lock_file_path)
213:                except Exception:
214:                    return

216:    def sleep(self, level: int = 1) -> None:
217:        free_bytes_before_sleep = torch.npu.mem_get_info()[0]
218:        # Save the buffers before level 2 sleep
219:        if level == 2:
220:            model = self.model_runner.model
221:            self._sleep_saved_buffers = {name: buffer.cpu().clone() for name, buffer in model.named_buffers()}
222:
223:        cleanup_enabled = getattr(get_ascend_config(), "enable_sleep_mode_extra_cleanup", False)
224:        if cleanup_enabled:
225:            self.sleep_wakeup_manager.sleep()
226:
227:        allocator = CaMemAllocator.get_instance()
228:        allocator.sleep(offload_tags=("weights",) if level == 1 else tuple())
229:        free_bytes_after_sleep, total = torch.npu.mem_get_info()
230:        freed_bytes = free_bytes_after_sleep - free_bytes_before_sleep
231:        used_bytes = total - free_bytes_after_sleep
232:        assert freed_bytes >= 0, "Memory usage increased after sleeping."
233:
234:        logger.info(
235:            "Sleep mode (level=%s) freed %.2f GiB memory, %.2f GiB memory is still in use.",
236:            level,
237:            freed_bytes / GiB_bytes,
238:            used_bytes / GiB_bytes,
239:        )

241:    def wake_up(self, tags: list[str] | None = None) -> None:
242:        nz_mode = get_ascend_config().weight_nz_mode
243:        if nz_mode:
244:            raise ValueError(
245:                "FRACTAL_NZ mode is enabled. This may cause model parameter precision issues "
246:                "in the RL scenarios. Please set weight_nz_mode=0 via --additional-config."
247:            )
248:        allocator = CaMemAllocator.get_instance()
249:        allocator.wake_up(tags=tags)
250:
251:        hidden_size = self.vllm_config.model_config.hf_text_config.hidden_size
252:        model = self.model_runner.model
253:        if self.vllm_config.quant_config is None and (tags is None or "weights" in tags):
254:            for name, param in model.named_parameters():
255:                if "w2_weight" in name and param.shape[2] == hidden_size:
256:                    parts = name.split(".")
257:                    param_name = parts[-1]
258:                    parent_module = model.get_submodule(".".join(parts[:-1]))
259:
260:                    w2_data = param.transpose(1, 2)
261:                    w2_data = torch.nn.Parameter(w2_data, requires_grad=False)
262:                    setattr(parent_module, param_name, w2_data)
263:                elif "w13_weight" in name and param.shape[1] == hidden_size:
264:                    parts = name.split(".")
265:                    param_name = parts[-1]
266:                    parent_module = model.get_submodule(".".join(parts[:-1]))
267:
268:                    w13_data = param.transpose(1, 2)
269:                    w13_data = torch.nn.Parameter(w13_data, requires_grad=False)
270:                    setattr(parent_module, param_name, w13_data)
271:
272:        # Restore the buffers after level 2 sleep
273:        if len(self._sleep_saved_buffers):
274:            for name, buffer in model.named_buffers():
275:                if name in self._sleep_saved_buffers:
276:                    buffer.data.copy_(self._sleep_saved_buffers[name].data)
277:            self._sleep_saved_buffers = {}
278:        cleanup_enabled = getattr(get_ascend_config(), "enable_sleep_mode_extra_cleanup", False)
279:        if cleanup_enabled:
280:            self.sleep_wakeup_manager.wakeup(tags)

303:    def start_weight_update(self, is_checkpoint_format: bool = True) -> None:
304:        """Begin a new weight update; prepares the model for layerwise reload."""
305:        self._check_weight_transfer_engine()
306:
307:        if self._weight_update_active:
308:            raise RuntimeError(
309:                "start_weight_update called while a weight update is already active. Call finish_weight_update first."
310:            )
311:
312:        self._check_nz_disabled()
313:
314:        if is_checkpoint_format:
315:            from vllm.model_executor.model_loader.reload import initialize_layerwise_reload
316:
317:            model = self.model_runner.model
318:            with torch.device(self.device):
319:                initialize_layerwise_reload(model)
320:
321:        self._is_checkpoint_format = is_checkpoint_format
322:        self._weight_update_active = True

324:    def update_weights(self, update_info: dict) -> None:
325:        """Receive a chunk of weights from the trainer and load them in place."""
326:        self._check_weight_transfer_engine()
327:        assert self.weight_transfer_engine is not None
328:
329:        typed_update_info = self.weight_transfer_engine.parse_update_info(update_info)
330:        model = self.model_runner.model
331:
332:        # state machine driven by start/finish.
333:        if not self._weight_update_active:
334:            raise RuntimeError("start_weight_update must be called before update_weights.")
335:
336:        with torch.device(self.device):
337:            if self._is_checkpoint_format:
338:                self.weight_transfer_engine.receive_weights(
339:                    typed_update_info,
340:                    load_weights=model.load_weights,
341:                )
342:            else:
343:
344:                def load_weights_direct(weights: list[tuple[str, torch.Tensor]]) -> None:
345:                    with torch.no_grad():
346:                        for name, weight in weights:
347:                            param = model.get_parameter(name)
348:                            param.copy_(weight)
349:
350:                self.weight_transfer_engine.receive_weights(
351:                    typed_update_info,
352:                    load_weights=load_weights_direct,
353:                )
354:
355:        # HCCL broadcast / packed paths are asynchronous.
356:        # Sync so the next step uses the new weights.
357:        torch.npu.synchronize()

359:    def finish_weight_update(self) -> None:
360:        """Finish the current weight update; runs layerwise postprocessing."""
361:        self._check_weight_transfer_engine()
362:
363:        if not self._weight_update_active:
364:            raise RuntimeError("start_weight_update must be called before finish_weight_update.")

366:        if self._is_checkpoint_format:
367:            from vllm.model_executor.model_loader.reload import finalize_layerwise_reload
368:
369:            model = self.model_runner.model
370:            with torch.device(self.device):
371:                finalize_layerwise_reload(model, self.model_config)
372:
373:        self._weight_update_active = False
374:        self._is_checkpoint_format = True

395:    def _init_device(self):
396:        if not vllm_version_is("0.23.0"):
397:            # vLLM v0.24.0 (PR #45026) removed automatic per-process device
398:            # isolation for DP workers. Mirror gpu_worker.py::init_device:
399:            # shift self.local_rank by dp_local_rank * tp_pp_world_size so
400:            # that each DP group binds to a distinct set of NPUs.
401:            parallel_config = self.parallel_config
402:            if (
403:                parallel_config.distributed_executor_backend not in ("ray", "external_launcher")
404:                and parallel_config.data_parallel_backend != "ray"
405:                and parallel_config.nnodes_within_dp == 1
406:                # vllm-ascend: when the user pre-shards devices via
407:                # --device-ids (which becomes assigned_physical_gpu_ids),
408:                # each child process already binds to its own NPU(s); the
409:                # DP local_rank shift below would push local_rank past the
410:                # length of the per-rank device list and trip the assert
411:                # in this same method. Skip the shift in that case.
412:                and parallel_config.assigned_physical_gpu_ids is None
413:            ):
414:                dp_local_rank = parallel_config.data_parallel_rank_local
415:                if dp_local_rank is None:
416:                    dp_local_rank = parallel_config.data_parallel_index
417:                tp_pp_world_size = parallel_config.pipeline_parallel_size * parallel_config.tensor_parallel_size
418:                self.local_rank += dp_local_rank * tp_pp_world_size
419:
420:            # Publish the logical-to-physical mapping for topology queries.
421:            assigned_physical_gpu_ids = parallel_config.assigned_physical_gpu_ids
422:            if assigned_physical_gpu_ids is not None:
423:                from vllm.platforms.interface import set_assigned_physical_gpu_ids
424:
425:                set_assigned_physical_gpu_ids(assigned_physical_gpu_ids)
426:                assert self.local_rank < len(assigned_physical_gpu_ids), (
427:                    f"local_rank {self.local_rank} is out of bounds for "
428:                    f"assigned_physical_gpu_ids {assigned_physical_gpu_ids}"
429:                )
430:                if parallel_config.distributed_executor_backend not in ("ray", "external_launcher"):
431:                    assert parallel_config.local_world_size <= len(assigned_physical_gpu_ids), (
432:                        f"local_world_size ({parallel_config.local_world_size}) "
433:                        f"exceeds assigned_physical_gpu_ids count "
434:                        f"({len(assigned_physical_gpu_ids)})"
435:                    )
436:            else:
437:                visible_device_count = torch.npu.device_count() if torch.npu.is_available() else 0
438:                assert self.local_rank < visible_device_count, (
439:                    f"DP adjusted local rank {self.local_rank} is out of bounds for {visible_device_count} devices."
440:                )
441:
442:            visible_device_index = current_platform.logical_device_id_to_visible_device_id(self.local_rank)
443:            device = torch.device(f"{current_platform.device_type}:{visible_device_index}")
444:        else:
445:            device = torch.device(f"npu:{self.local_rank}")
446:
447:        torch.npu.set_device(device)
448:
449:        # Import _inductor for graph mode execution with triton
450:        # This lazy import avoids torch_npu re-initialization in patch
451:        # Note that this should be imported after torch.npu.set_device
452:        # to avoid repeated set_device in extra processes
453:        from vllm.triton_utils import HAS_TRITON
454:
455:        if HAS_TRITON:
456:            import torch_npu._inductor  # noqa: F401
457:
458:        gc.collect()
459:        torch.npu.empty_cache()
460:
461:        if get_ascend_device_type() == AscendDeviceType.A5:
462:            setup_ascend_local_comm_res(self.local_rank, self.vllm_config.kv_transfer_config)
463:
464:        # take current memory snapshot
465:        if vllm_version_is("0.23.0"):
466:            self.init_snapshot = MemorySnapshot()
467:        else:
468:            self.init_snapshot = MemorySnapshot(device=device)
469:        self.requested_memory = self.init_snapshot.total_memory * self.cache_config.gpu_memory_utilization
470:        if self.init_snapshot.free_memory < self.requested_memory:
471:            GiB = lambda b: round(b / GiB_bytes, 2)
472:            raise ValueError(
473:                f"Free memory on device "
474:                f"({GiB(self.init_snapshot.free_memory)}/"
475:                f"{GiB(self.init_snapshot.total_memory)} GiB) on startup "
476:                f"is less than desired GPU memory utilization "
477:                f"({self.cache_config.gpu_memory_utilization}, "
478:                f"{GiB(self.requested_memory)} GiB). Decrease GPU memory "
479:                f"utilization or reduce GPU memory used by other processes."
480:            )
481:
482:        if (
483:            self.parallel_config.data_parallel_size > 1
484:            and self.parallel_config.data_parallel_size_local > 0
485:            and self.parallel_config.distributed_executor_backend not in ["ray", "external_launcher"]
486:            and self.vllm_config.parallel_config.data_parallel_backend != "ray"
487:            and self.vllm_config.parallel_config.nnodes_within_dp == 1
488:        ):
489:            visible_device_count = torch.npu.device_count() if torch.npu.is_available() else 0
490:            assert self.parallel_config.local_world_size <= visible_device_count, (
491:                f"local_world_size ({self.parallel_config.local_world_size}) must "
492:                f"be less than or equal to the number of visible devices "
493:                f"({visible_device_count})."
494:            )
495:
496:        # Initialize the distributed environment.
497:        self._init_worker_distributed_environment()
498:        # Set random seed.
499:        set_random_seed(self.model_config.seed)
500:        # Initialize device properties used by triton kernels.
501:        init_device_properties_triton()
502:
503:        return device

526:    @torch.inference_mode()
527:    def determine_available_memory(self) -> int:
528:        """Profiles the peak memory usage of the model to determine how much
529:        memory can be used for KV cache without OOMs.
530:
531:        The engine will first conduct a profiling of the existing memory usage.
532:        Then, it calculates the free memory that can be used for KV cache in
533:        bytes.
534:        """
535:        GiB = lambda b: b / GiB_bytes
536:
537:        # Fast path: user has explicitly specified KV cache size via
538:        # --kv-cache-memory. Still run profile_run() to compile the model,
539:        # but skip the memory profiling calculation entirely.
540:        if kv_cache_memory_bytes := self.cache_config.kv_cache_memory_bytes:
541:            self.model_runner.profile_run()
542:            logger.info(
543:                "Initial free memory %.2f GiB, reserved %.2f GiB for KV Cache "
544:                "as specified by kv_cache_memory_bytes, skipping memory profiling. "
545:                "This does not respect the gpu_memory_utilization config. "
546:                "Only use kv_cache_memory_bytes when you want manual control of "
547:                "KV cache memory size. If OOM'ed, check the difference of initial "
548:                "free memory between the current run and the previous run where "
549:                "kv_cache_memory_bytes is suggested and update it correspondingly.",
550:                GiB(self.init_snapshot.free_memory),
551:                GiB(kv_cache_memory_bytes),
552:            )
553:            return kv_cache_memory_bytes
554:
555:        # Execute a forward pass with dummy inputs to profile the memory usage
556:        # of the model.
557:        with memory_profiling(
558:            self.init_snapshot,
559:            weights_memory=int(self.model_runner.model_memory_usage),
560:        ) as profile_result:
561:            self.model_runner.profile_run()
562:
563:            # Record torch peak INSIDE the context and BEFORE graph capture,
564:            # so that graph pool allocations don't inflate the activation peak.
565:            # The memory_profiling context will also compute torch_peak_increase
566:            # on exit, but we override it below with this pre-graph value.
567:            profile_torch_peak = torch.npu.memory_stats(self.device).get("allocated_bytes.all.peak", 0)
568:
569:        # Override torch_peak_increase with the pre-graph-capture value to
570:        # avoid double-counting graph pool memory as activation memory.
571:        profile_result.torch_peak_increase = profile_torch_peak - profile_result.before_profile.torch_peak
572:        profile_result.non_kv_cache_memory = (
573:            profile_result.non_torch_increase + profile_result.torch_peak_increase + profile_result.weights_memory
574:        )
575:
576:        # Save per-category memory for use in compile_or_warm_up_model() (step 5).
577:        self.peak_activation_memory = profile_result.torch_peak_increase
578:        self.non_torch_memory = profile_result.non_torch_increase
579:
580:        free_gpu_memory = profile_result.after_profile.free_memory
581:        assert self.init_snapshot.free_memory > free_gpu_memory, (
582:            "Error in memory profiling. "
583:            f"Initial free memory {GiB(self.init_snapshot.free_memory)} GiB, "
584:            f"current free memory {GiB(free_gpu_memory)} GiB. "
585:            "This happens when other processes sharing the same container "
586:            "release GPU memory while vLLM is profiling during initialization. "
587:            "To fix this, ensure consistent GPU memory allocation or "
588:            "isolate vLLM in its own container."
589:        )
590:        self.available_kv_cache_memory_bytes = self.requested_memory - profile_result.non_kv_cache_memory
591:
592:        logger.debug(profile_result)
593:        logger.info_once(
594:            "Available KV cache memory: %.2f GiB", GiB(self.available_kv_cache_memory_bytes), scope="local"
595:        )
596:
597:        return int(self.available_kv_cache_memory_bytes)

611:    def execute_model(
612:        self,
613:        scheduler_output: "SchedulerOutput",
614:    ) -> ModelRunnerOutput | AsyncModelRunnerOutput | None:
615:        self.log_memory_stats()
616:        # enable msMonitor to monitor the performance of vllm-ascend
617:        if get_ascend_config().msmonitor_use_daemon:
618:            dp.step()
619:
620:        if self._pp_send_work:
621:            for handle in self._pp_send_work:
622:                handle.wait()
623:            self._pp_send_work = []
624:
625:        intermediate_tensors = None
626:        forward_pass = scheduler_output.total_num_scheduled_tokens > 0
627:        if forward_pass and not get_pp_group().is_first_rank:
628:            # If flashcomm1 is used, this all_gather_group parameter needs to be removed, otherwise
629:            # it will conflict with the all-gather operation in flashcomm1.
630:            if enable_sp():
631:                all_gather_group = None
632:            else:
633:                all_gather_group = get_tp_group()
634:            tensor_dict, comm_handles, comm_postprocess = get_pp_group().irecv_tensor_dict(
635:                all_gather_group=all_gather_group
636:            )
637:            assert tensor_dict is not None
638:            intermediate_tensors = AsyncIntermediateTensors(
639:                tensor_dict,
640:                comm_handles=comm_handles,
641:                comm_postprocess=comm_postprocess,
642:            )
643:
644:        if self.profiler is not None:
645:            self.profiler.step()
646:
647:        output = self.model_runner.execute_model(scheduler_output, intermediate_tensors)
648:        if isinstance(output, (ModelRunnerOutput, AsyncModelRunnerOutput, NoneType)):
649:            return output
650:
651:        assert isinstance(output, IntermediateTensors)
652:        parallel_config = self.vllm_config.parallel_config
653:        assert parallel_config.distributed_executor_backend != ("external_launcher") and not get_pp_group().is_last_rank
654:        # If flashcomm1 is used, this all_gather_group parameter needs to be removed, otherwise
655:        # it will conflict with the all-gather operation in flashcomm1.
656:        if enable_sp():
657:            all_gather_group = None
658:        else:
659:            all_gather_group = get_tp_group()
660:        self._pp_send_work = get_pp_group().isend_tensor_dict(
661:            output.tensors,
662:            all_gather_group=all_gather_group,
663:        )
664:
665:        kv_connector_output = output.kv_connector_output
666:        if not kv_connector_output:
667:            return None
668:
669:        # In case of PP with kv transfer, we need to pass through the
670:        # kv_connector_output
671:        if not kv_connector_output.finished_sending and not kv_connector_output.finished_recving:
672:            return EMPTY_MODEL_RUNNER_OUTPUT
673:        output = copy.copy(EMPTY_MODEL_RUNNER_OUTPUT)
674:        output.kv_connector_output = kv_connector_output
675:        return output

706:    def compile_or_warm_up_model(self) -> CompilationTimes:
707:        # Note: need to adapt for graph mode.
708:        warmup_sizes = (self.vllm_config.compilation_config.compile_sizes or []).copy()
709:        if not self.model_config.enforce_eager:
710:            cg_capture_sizes: list[int] = []
711:            if self.vllm_config.compilation_config.cudagraph_mode != CUDAGraphMode.NONE:
712:                cg_sizes = self.vllm_config.compilation_config.cudagraph_capture_sizes
713:                cg_capture_sizes = [] if cg_sizes is None else cg_sizes
714:                warmup_sizes = [x for x in warmup_sizes if x not in cg_capture_sizes]
715:
716:            compile_ranges = self.vllm_config.compilation_config.get_compile_ranges()
717:            # For each compile_range, if none of the batch sizes
718:            # in warmup_sizes or cudagraph_capture_sizes are in the range,
719:            # add the end of the range to ensure compilation/warmup.
720:            all_sizes = set(cg_capture_sizes)
721:            all_sizes.update([x for x in warmup_sizes if isinstance(x, int)])
722:            for compile_range in compile_ranges:
723:                if not any(x in compile_range for x in all_sizes):
724:                    warmup_sizes.append(compile_range.end)
725:
726:        for size in sorted(warmup_sizes, reverse=True):
727:            logger.info("Compile and warming up model for size %d", size)
728:            self.model_runner._dummy_run(size)
729:
730:        npugraph_memory_bytes = 0
731:        if not self.model_config.enforce_eager:
732:            npugraph_memory_bytes = self.model_runner.capture_model()
733:
734:        # Suggest an optimal --kv-cache-memory value for future runs.
735:        # Only emitted when we ran full profiling (kv_cache_memory_bytes was not
736:        # pre-specified) so that peak_activation_memory etc. are available.
737:        # non_kv_memory already includes NPU graph memory, so the suggestion
738:        # accounts for all measured memory categories. A 150 MiB buffer is kept
739:        # because memory_profiling may slightly underestimate non-torch
740:        # allocations (ACL context, HCCL buffers, driver layer, etc.).
741:        if self.cache_config.kv_cache_memory_bytes is None and hasattr(self, "peak_activation_memory"):
742:            redundancy_buffer = 150 * (1 << 20)  # 150 MiB safety margin
743:            non_kv_memory = (
744:                self.model_runner.model_memory_usage
745:                + self.peak_activation_memory
746:                + self.non_torch_memory
747:                + npugraph_memory_bytes
748:            )
749:            self.npugraph_memory_bytes = npugraph_memory_bytes
750:            suggested_to_requested = int(self.requested_memory) - non_kv_memory - redundancy_buffer
751:            suggested_to_gpu_limit = int(self.init_snapshot.free_memory) - non_kv_memory - redundancy_buffer
752:            msg = (
753:                f"Free memory on device "
754:                f"({format_gib(self.init_snapshot.free_memory)}/"
755:                f"{format_gib(self.init_snapshot.total_memory)} GiB) on startup. "
756:                f"Desired GPU memory utilization is "
757:                f"({self.cache_config.gpu_memory_utilization}, "
758:                f"{format_gib(self.requested_memory)} GiB). "
759:                f"Actual usage: {format_gib(self.model_runner.model_memory_usage)} GiB "
760:                f"for weights, {format_gib(self.peak_activation_memory)} GiB for peak "
761:                f"activation, {format_gib(self.non_torch_memory)} GiB for non-torch "
762:                f"memory, {format_gib(npugraph_memory_bytes)} GiB for NPU graph memory. "
763:                f"Replace gpu_memory_utilization with "
764:                f"`--kv-cache-memory={suggested_to_requested}` "
765:                f"({format_gib(suggested_to_requested)} GiB) to fit into requested "
766:                f"memory, or `--kv-cache-memory={suggested_to_gpu_limit}` "
767:                f"({format_gib(suggested_to_gpu_limit)} GiB) to fully utilize NPU "
768:                f"free memory. Current KV cache memory: "
769:                f"{format_gib(self.available_kv_cache_memory_bytes)} GiB."
770:            )
771:            logger.info(msg)
772:
773:        # Call ATB matmul to warm up; otherwise, the first operation (ReshapeAndCache)
774:        # may cause performance degradation at runtime.
775:        if get_ascend_device_type() != AscendDeviceType.A5:
776:            self._warm_up_atb()
777:        # Bind after warmup so hot allocations are already materialized on the
778:        # worker process before migratepages/taskset run.
779:        if get_ascend_config().enable_cpu_binding:
780:            try:
781:                bind_cpus(self.local_rank)
782:            except Exception as e:
783:                logger.warning("Bind cpus failed in rank%s: %s Skip binding cpu.", self.local_rank, e)
784:        # Reset the seed to ensure that the random state is not affected by
785:        # the model initialization and profiling.
786:        set_random_seed(self.model_config.seed)
787:        return CompilationTimes(
788:            language_model=self.vllm_config.compilation_config.compilation_time,
789:            # `encoder_compilation_time` was added after v0.19.1 (vLLM #39240); fall
790:            # back to 0.0 so the older release still constructs CompilationTimes.
791:            encoder=getattr(
792:                self.vllm_config.compilation_config,
793:                "encoder_compilation_time",
794:                0.0,
795:            ),
796:        )

807:    @torch.inference_mode()
808:    def profile_prefill_latency(self, num_tokens: int) -> float:
809:        """
810:        Profile prefill latency for a given number of tokens.
811:
812:        This runs a real model forward pass and measures the execution time.
813:        Used for profiling-based dynamic chunk sizing.
814:
815:        In PP (Pipeline Parallelism) mode:
816:        - All workers execute the forward pass to stay synchronized
817:        - Only the timing from PP0 (first rank) is meaningful for scheduling
818:        - PP0 includes all the pipeline stages' latency when using async scheduling
819:
820:        Args:
821:            num_tokens: Number of tokens to profile
822:
823:        Returns:
824:            Latency in milliseconds
825:        """
826:        import time
827:
828:        # Clamp to valid range
829:        num_tokens = min(num_tokens, self.scheduler_config.max_num_batched_tokens)
830:        num_tokens = max(num_tokens, 1)
831:
832:        # Synchronize all devices before timing
833:        # This ensures clean measurement in PP/TP scenarios
834:        torch.npu.synchronize()
835:
836:        # In PP mode, we still run on all ranks to keep them synchronized
837:        # but only the first rank's timing is used for scheduling decisions
838:        is_first_pp_rank = get_pp_group().is_first_rank
839:
840:        start = time.perf_counter()
841:
842:        # Run real model forward with force_attention=True
843:        # This ensures attention is actually executed, not skipped.
844:        # Without force_attention, attn_metadata may be None and attention
845:        # won't run, making profiling results inaccurate.
846:        # _dummy_run handles PP internally (intermediate tensors, etc.)
847:        self.model_runner._dummy_run(
848:            num_tokens=num_tokens,
849:            force_attention=True,  # Critical: ensure attention is executed
850:            profile_cpp=True,
851:        )
852:
853:        # Synchronize after forward to ensure NPU operations complete
854:        torch.npu.synchronize()
855:
856:        latency_ms = (time.perf_counter() - start) * 1000
857:
858:        # Log for debugging in PP mode
859:        if not is_first_pp_rank:
860:            if logger.isEnabledFor(logging.DEBUG):
861:                logger.debug(
862:                    "[ProfilingChunk] PP rank %s: profiled %s tokens, latency=%.2f ms (not used)",
863:                    get_pp_group().rank_in_group,
864:                    num_tokens,
865:                    latency_ms,
866:                )
867:
868:        return latency_ms

54:    def __init__(self, expected_finished_count: int):
55:        # Complete transfer tracker. Used to track finished requests
56:        # [req_id -> n_remaining_workers]
57:        self._recv_remaining_count = dict[str, int]()
58:        self._send_remaining_count = dict[str, int]()
59:        self._expected_finished_count = expected_finished_count

61:    @classmethod
62:    def from_connector(cls, connector: "KVConnectorBase", world_size: int):

65:    def aggregate(
66:        self, outputs: list[ModelRunnerOutput | None], output_rank: int = 0
67:    ) -> ModelRunnerOutput | None:
68:        if not outputs[output_rank]:
69:            return None
70:
71:        # Aggregate kv_connector_output from all workers
72:
73:        def update_finished_set(
74:            req_ids: set[str] | None,
75:            remaining_count_dict: dict[str, int],
76:            finished_set: set[str],
77:        ) -> None:
78:            for req_id in req_ids or ():
79:                remaining_count = remaining_count_dict.get(
80:                    req_id, self._expected_finished_count
81:                )
82:                remaining_count_dict[req_id] = remaining_count - 1
83:                if remaining_count_dict[req_id] == 0:
84:                    finished_set.add(req_id)
85:                    del remaining_count_dict[req_id]
86:
87:        finished_sending = set[str]()
88:        finished_recving = set[str]()
89:        aggregated_kv_connector_stats = None
90:        aggregated_kv_connector_worker_meta = None
91:        combined_kv_cache_events = None
92:        invalid_block_ids = set[int]()
93:        for model_runner_output in outputs:
94:            assert model_runner_output is not None
95:            kv_output = model_runner_output.kv_connector_output
96:            if not kv_output:
97:                continue
98:            # Allow the worker to dynamically update the expected number of
99:            # finished sending/recving for new requests.
100:            if (
101:                kv_output.expected_finished_count > 0
102:                and kv_output.expected_finished_count != self._expected_finished_count
103:            ):
104:                logger.debug(
105:                    "Expected finished requests updated from %d to %d",
106:                    self._expected_finished_count,
107:                    kv_output.expected_finished_count,
108:                )
109:                self._expected_finished_count = kv_output.expected_finished_count
110:
111:            update_finished_set(
112:                kv_output.finished_sending, self._send_remaining_count, finished_sending
113:            )
114:            update_finished_set(
115:                kv_output.finished_recving, self._recv_remaining_count, finished_recving
116:            )
117:
118:            # Aggregate kv_connector_stats from all workers.
119:            if aggregated_kv_connector_stats is None:
120:                # Use the first worker's kv_connector_stats as accumulator.
121:                aggregated_kv_connector_stats = kv_output.kv_connector_stats
122:            elif kv_connector_stats := kv_output.kv_connector_stats:
123:                assert isinstance(
124:                    aggregated_kv_connector_stats, type(kv_connector_stats)
125:                )
126:                aggregated_kv_connector_stats = aggregated_kv_connector_stats.aggregate(
127:                    kv_connector_stats
128:                )
129:
130:            # Aggregate kv_connector_worker_meta from all workers.
131:            if aggregated_kv_connector_worker_meta is None:
132:                # Use the first worker's kv_connector_worker_meta as accumulator.
133:                aggregated_kv_connector_worker_meta = kv_output.kv_connector_worker_meta
134:            elif kv_connector_worker_meta := kv_output.kv_connector_worker_meta:
135:                aggregated_kv_connector_worker_meta = (
136:                    aggregated_kv_connector_worker_meta.aggregate(
137:                        kv_connector_worker_meta
138:                    )
139:                )
140:
141:            # Combine kv_cache_events from all workers.
142:            if combined_kv_cache_events is None:
143:                # Use the first worker's kv_cache events as start event list.
144:                combined_kv_cache_events = kv_output.kv_cache_events
145:            elif kv_cache_events := kv_output.kv_cache_events:
146:                assert isinstance(
147:                    combined_kv_cache_events,
148:                    type(kv_cache_events),
149:                )
150:                worker_kv_cache_events = kv_cache_events.get_all_events()
151:                combined_kv_cache_events.add_events(worker_kv_cache_events)
152:                combined_kv_cache_events.increment_workers(1)
153:
154:            invalid_block_ids |= kv_output.invalid_block_ids
155:
156:        # select output of the worker specified by output_rank
157:        output = outputs[output_rank]
158:
159:        assert output is not None
160:        output.kv_connector_output = KVConnectorOutput(
161:            finished_sending=finished_sending or None,
162:            finished_recving=finished_recving or None,
163:            kv_connector_stats=aggregated_kv_connector_stats or None,
164:            kv_cache_events=combined_kv_cache_events or None,
165:            kv_connector_worker_meta=aggregated_kv_connector_worker_meta or None,
166:            invalid_block_ids=invalid_block_ids,
167:            expected_finished_count=self._expected_finished_count,
168:        )
169:
170:        return output

221:def kv_postprocess_blksize_on_receive(cache, indices, block_size_ratio):
222:    """
223:    Transforms the layout of received KV cache blocks to the local block_size.
224:    (Only works for local blocksize > remote blocksize)
225:
226:    example:
227:    local blocksize = 16 tokens, remote blocksize = 4 tokens
228:    local block[0] = remote block[0, 1, 2, 3]
229:    remote is |h0-b0|h1-b0|h2-b0|h3-b0|h0-b1|h1-b1|h2-b1|h3-b1|...
230:    local is  |h0-b0..................|h1-b0..................|...
231:    permute is to:
232:    1. view => view remote as n_blocks * remote_shape(H,remoteN,D)
233:    2. permute => (H, nblocks, remoteN, D)
234:    3. flatten => (H, localN, D)
235:    """
236:    blocks_to_update = cache.index_select(0, indices)
237:    # use physical order
238:    blocks_to_update = blocks_to_update.permute(0, 2, 1, 3)
239:    n_kv_heads, block_size, head_size = blocks_to_update.shape[1:]
240:    remote_block_size = block_size // block_size_ratio
241:    n_blocks = block_size_ratio
242:
243:    permuted_blocks = (
244:        blocks_to_update.reshape(-1, n_blocks, n_kv_heads, remote_block_size, head_size)
245:        .permute(0, 2, 1, 3, 4)
246:        .flatten(2, 3)
247:    )
248:    permuted_blocks = permuted_blocks.permute(0, 2, 1, 3)
249:    cache.index_copy_(0, indices, permuted_blocks)

252:def kv_postprocess_layout_on_receive(cache, indices):
253:    """Transforms the layout of received KV cache blocks to the local format.
254:
255:    This method corrects layout mismatches from direct memory copies by
256:    permuting the tensor dimensions.
257:
258:    - **Source Layout:** `[num_blocks, n_kv_head, block_size, head_dim]`
259:    - **Target Layout:** `[num_blocks, block_size, n_kv_head, head_dim]`
260:
261:    Implementation:
262:    - x = blocks_to_update.reshape(src_shape) # view local kv with sender layout
263:    - permuted_blocks = x.permute(*inv_order) # transpose n_kv_heads, block_size
264:    - cache.index_copy_(0, indices, permuted_blocks) # copy permuted kv back
265:
266:    """
267:    blocks_to_update = cache.index_select(0, indices)
268:    target_shape = list(blocks_to_update.shape)
269:    target_shape[0] = -1
270:    inv_order = [0, 2, 1, 3]
271:    src_shape = tuple(target_shape[i] for i in inv_order)
272:    blocks_to_update = cache.index_select(0, indices)
273:    permuted_blocks = blocks_to_update.reshape(src_shape).permute(*inv_order)
274:    cache.index_copy_(0, indices, permuted_blocks)

278:    """
279:    Transforms the layout of received KV cache to the local block_size and HND.
280:    (Only works for local blocksize > remote blocksize)
281:
282:    prefill is HND, smaller block_size

295:    )
296:    cache.index_copy_(0, indices, permuted_blocks)

516:    def tp_ratio(self, remote_tp_size: int) -> int:
517:        """Calculate the tensor parallel ratio between local and remote TP.
518:
519:        Positive when local_tp >= remote_tp (local workers read from the
520:        same remote worker in groups of size ``tp_ratio``).  Negative when
521:        remote_tp > local_tp (ratio is flipped).
522:        """
523:        if self.tp_size >= remote_tp_size:
524:            assert self.tp_size % remote_tp_size == 0, (
525:                f"Local tensor parallel size {self.tp_size} is not divisible "
526:                f"by remote tensor parallel size {remote_tp_size}."
527:            )
528:            return self.tp_size // remote_tp_size
529:        assert remote_tp_size % self.tp_size == 0, (
530:            f"Remote tensor parallel size {remote_tp_size} is not divisible "
531:            f"by local tensor parallel size {self.tp_size}."
532:        )
533:        return -(remote_tp_size // self.tp_size)
534:
535:    def block_size_ratio(self, remote_block_size: int) -> int:
536:        """Calculate the block size ratio between local and remote."""
537:        assert self.block_size % remote_block_size == 0, (
538:            f"Local block size {self.block_size} is not divisible "
539:            f"by remote block size {remote_block_size} or vice versa."
540:        )
541:        return self.block_size // remote_block_size

566:        """Pre-registration: compute which remote TP ranks to handshake with.
567:
568:        Pure math based on local/remote TP sizes — does not require
569:        the remote engine to be registered yet.
570:        """
571:        tp_ratio = self.tp_ratio(remote_tp_size)
572:        if tp_ratio > 0:
573:            return [self.tp_rank // tp_ratio]
574:        abs_ratio = -tp_ratio
575:        return [self.tp_rank * abs_ratio + i for i in range(abs_ratio)]

579:    ) -> list[int]:
580:        """Get the remote TP rank(s) that the current local TP rank will

592:    def get_transfer_cache_regions(
593:        self, cache: torch.Tensor, layer_spec: "KVCacheSpec"
594:    ) -> list[torch.Tensor] | torch.Tensor:
595:        """Return the cache tensor(s) to register as NIXL memory regions,
596:        also accounting for hybrid SSM models specificities.
597:        """
598:        if isinstance(layer_spec, MambaSpec):
599:            # Register the whole kv cache shared tensor, including
600:            # SSM/Conv.
601:            conv, ssm = cache
602:            return [conv]
603:
604:        # Check may be hacky but it's matching
605:        # `_update_hybrid_attention_mamba_layout`.
606:        if self.is_mamba and cache.shape[0] == 2:
607:            # When MAMBA is present, all backends are blocks first, so
608:            # that blocks can be shared between attention layers and mamba
609:            # layers.  Runner already adjusted strides for FlashAttn-like
610:            # backends so its num_blocks first.
611:            # Swap [2<>num_blocks] dims for hybrid SSM layout.
612:            cache = cache.transpose(0, 1)
613:
614:        # Regular case: backends like FA register K/V in separate regions
615:        return cache if self.split_k_and_v else [cache]
